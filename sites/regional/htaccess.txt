# =========================================================
#  {{domain}} – konfigurace Apache
#  (generováno z sites/regional/htaccess.txt)
# =========================================================

Options -Indexes

AddDefaultCharset UTF-8
DirectoryIndex index.html

# ---------------------------------------------------------
#  Přesměrování
# ---------------------------------------------------------
<IfModule mod_rewrite.c>
  RewriteEngine On

  # HTTP -> HTTPS a bez www -> www (jedním 301 skokem)
  RewriteCond %{HTTPS} !=on
  RewriteCond %{HTTP:X-Forwarded-Proto} !=https
  RewriteRule ^ https://{{domain}}%{REQUEST_URI} [R=301,L,NE]

  RewriteCond %{HTTP_HOST} !^www\. [NC]
  RewriteRule ^ https://{{domain}}%{REQUEST_URI} [R=301,L,NE]

  # Blokace skrytých souborů (.git, .env, ...)
  RewriteRule (^|/)\. - [F,L]

  # Původní stránky z WordPressu -> odpovídající sekce nového webu
{{legacyRedirects}}

  # Cokoliv dalšího ze starého webu, co už neexistuje (wp-*, feed, project/...) -> úvodní stránka
  RewriteCond %{REQUEST_FILENAME} !-f
  RewriteCond %{REQUEST_FILENAME} !-d
  RewriteRule ^.+$ / [R=301,L]
</IfModule>

# ---------------------------------------------------------
#  MIME typy
# ---------------------------------------------------------
<IfModule mod_mime.c>
  AddType image/webp            .webp
  AddType application/geo+json  .geojson
  AddType image/svg+xml         .svg
</IfModule>

# ---------------------------------------------------------
#  Komprese
# ---------------------------------------------------------
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/css text/xml
  AddOutputFilterByType DEFLATE application/javascript text/javascript
  AddOutputFilterByType DEFLATE application/json application/geo+json
  AddOutputFilterByType DEFLATE application/xml image/svg+xml
</IfModule>

# ---------------------------------------------------------
#  Cache v prohlížeči
# ---------------------------------------------------------
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresDefault                       "access plus 1 week"
  ExpiresByType text/html              "access plus 0 seconds"
  ExpiresByType application/xml        "access plus 1 day"
  ExpiresByType text/plain             "access plus 1 day"
  ExpiresByType application/geo+json   "access plus 1 day"
  ExpiresByType text/css               "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 year"
  ExpiresByType text/javascript        "access plus 1 year"
  ExpiresByType image/webp             "access plus 1 month"
  ExpiresByType image/png              "access plus 1 month"
  ExpiresByType image/jpeg             "access plus 1 month"
  ExpiresByType application/pdf        "access plus 1 month"
</IfModule>

<IfModule mod_headers.c>
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache"
  </FilesMatch>

  <If "%{REQUEST_URI} =~ m#^/assets/#">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </If>

  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), interest-cohort=()"
  Header always set Strict-Transport-Security "max-age=31536000" "expr=%{HTTPS} == 'on' || %{HTTP:X-Forwarded-Proto} == 'https'"

  Header unset X-Powered-By
</IfModule>
